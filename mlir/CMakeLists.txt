# Out-of-tree build of the milo dialect and milo-opt against an installed MLIR.
cmake_minimum_required(VERSION 3.20)
project(milo-mlir LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
find_package(MLIR REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
include_directories(${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS} ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
add_definitions(${LLVM_DEFINITIONS})

set(LLVM_TARGET_DEFINITIONS Milo.td)
mlir_tablegen(MiloDialect.h.inc -gen-dialect-decls -dialect=milo)
mlir_tablegen(MiloDialect.cpp.inc -gen-dialect-defs -dialect=milo)
mlir_tablegen(MiloTypes.h.inc -gen-typedef-decls -typedefs-dialect=milo)
mlir_tablegen(MiloTypes.cpp.inc -gen-typedef-defs -typedefs-dialect=milo)
mlir_tablegen(MiloOps.h.inc -gen-op-decls)
mlir_tablegen(MiloOps.cpp.inc -gen-op-defs)
mlir_tablegen(MiloEnums.h.inc -gen-enum-decls)
mlir_tablegen(MiloEnums.cpp.inc -gen-enum-defs)
add_public_tablegen_target(MiloIncGen)

add_executable(milo-opt milo-opt.cpp Milo.cpp Passes.cpp Scalar.cpp Drops.cpp Link.cpp Desugar.cpp)
add_dependencies(milo-opt MiloIncGen)
if(NOT LLVM_ENABLE_RTTI)
  target_compile_options(milo-opt PRIVATE -fno-rtti)
endif()
if(MILO_STATIC)
  # Shipped builds: no dylib dependency on the build machine's paths (Homebrew prefix, apt
  # dirs). Every static MLIR archive and every LLVM component; the linker's dead-stripping
  # drops what milo-opt never calls.
  file(GLOB MILO_MLIR_LIBS "${MLIR_INSTALL_PREFIX}/lib/libMLIR*.a")
  list(FILTER MILO_MLIR_LIBS EXCLUDE REGEX "CAPI")
  file(GLOB MILO_LLVM_LIBS "${LLVM_LIBRARY_DIR}/libLLVM*.a")
  find_library(MILO_ZSTD NAMES libzstd.a zstd REQUIRED)
  if(APPLE)
    target_link_libraries(milo-opt PRIVATE ${MILO_MLIR_LIBS} ${MILO_LLVM_LIBS} -Wl,-dead_strip z ${MILO_ZSTD} curses)
  else()
    target_link_libraries(milo-opt PRIVATE -Wl,--start-group ${MILO_MLIR_LIBS} -Wl,--end-group ${MILO_LLVM_LIBS} -Wl,--gc-sections z ${MILO_ZSTD} pthread dl)
  endif()
else()
  target_link_libraries(milo-opt PRIVATE MLIR LLVM)
endif()
